async function apiCall(page, method, path, body) {
  return page.evaluate(async ([m, p, b]) => {
    const opts = { method: m, headers: { 'Content-Type': 'application/json' } };
    if (b) opts.body = JSON.stringify(b);
    const res = await fetch(p, opts);
    if (res.status === 204) return null;
    const data = await res.json();
    if (!res.ok) { const err = new Error(data.error || 'http_' + res.status); err.status = res.status; err.data = data; throw err; }
    return data;
  }, [method, path, body]);
}

test.describe('Recipes tab — usage_pct persistence (Phase 999.2)', () => {
  let seededRecipeId = null;
  let seededFromExisting = false; // true if we reused an existing recipe rather than POSTing

  // Local apiCall — uses the page.evaluate fetch wrapper (matches inventory.spec.js convention).
  async function apiCall(page, method, path, body) {
    return page.evaluate(async ([m, p, b]) => {
      const opts = { method: m, headers: { 'Content-Type': 'application/json' } };
      if (b) opts.body = JSON.stringify(b);
      const res = await fetch(p, opts);
      if (res.status === 204) return null;
      const data = await res.json();
      if (!res.ok) { const err = new Error(data.error || 'http_' + res.status); err.status = res.status; err.data = data; throw err; }
      return data;
    }, [method, path, body]);
  }

  test.beforeEach(async ({ page }) => {
    // Login + land on inventory so the cookie session is established for apiCall fetches.
    await page.goto(BASE + '/login.html');
    await page.fill('input[type="email"]', ADMIN_EMAIL);
    await page.fill('input[type="password"]', ADMIN_PASSWORD);
    await page.click('button.btn');
    await page.waitForURL(url => !url.pathname.includes('login'));
    await page.goto(BASE + '/inventory.html');
    await page.waitForLoadState('networkidle');

    // Try to reuse an existing recipe first — the persistence test only needs A recipe to drag.
    let recipes = [];
    try {
      const resp = await apiCall(page, 'GET', '/api/v1/inventory/recipes');
      recipes = (resp && resp.ingredients) || [];
    } catch (_) { /* empty DB — fall through to seed */ }

    const firstExisting = recipes.find(ing => (ing.recipes || []).length > 0);
    if (firstExisting) {
      // Reuse — record id so we DO NOT delete it in afterEach (it predated this test run).
      seededRecipeId = firstExisting.recipes[0].id;
      seededFromExisting = true;
      return;
    }

    // No recipes exist — seed one. Need a menu_item AND a purchase_item.
    let menuItems = [];
    try {
      const m = await apiCall(page, 'GET', '/api/v1/inventory/menu-items?since=90');
      menuItems = (m && m.menu_items) || m || [];
    } catch (_) { /* swallow */ }
    if (menuItems.length === 0) {
      test.skip(true, 'no menu_items in test DB — Toast ingest has not run');
      return;
    }
    // For purchase_item, scan all ingredients (with or without recipes attached).
    let purchaseItemId = null;
    for (const ing of recipes) {
      if (ing.purchase_item_id) { purchaseItemId = ing.purchase_item_id; break; }
    }
    if (!purchaseItemId) {
      // ingredients list was empty — try the broader purchase_items endpoint if available,
      // otherwise skip with a clear environment-gap message.
      test.skip(true, 'no purchase_items available — confirm a receipt via the Purchases tab to populate the test DB');
      return;
    }

    const created = await apiCall(page, 'POST', '/api/v1/inventory/recipes', {
      menu_item_id: menuItems[0].id,
      purchase_item_id: purchaseItemId,
      usage_pct: 25,
    });
    seededRecipeId = created && created.id;
    seededFromExisting = false;
    if (!seededRecipeId) {
      test.skip(true, 'recipe seed POST returned no id — see network for failure');
    }
  });

  test.afterEach(async ({ page }) => {
    // Only clean up recipes WE created (avoid clobbering pre-existing fixtures).
    if (seededRecipeId && !seededFromExisting) {
      try { await apiCall(page, 'DELETE', '/api/v1/inventory/recipes/' + seededRecipeId); } catch (_) { /* swallow */ }
    }
    seededRecipeId = null;
    seededFromExisting = false;
  });

  test('usage_pct survives tab switch and reopen — Recipes tab', async ({ page }) => {
    // CLAUDE.md persistence rule: every new field type / data entry feature MUST have
    // a back-and-reopen regression test. This test seeds its own recipe in beforeEach
    // so it ALWAYS exercises the round-trip — a perpetually-skipping test does not
    // fulfill the regression intent (per planner revision).

    await page.goto(BASE + '/inventory.html#tab=4');
    await page.waitForLoadState('networkidle');

    const rows = page.locator('.recipe-ingredient-row');
    const count = await rows.count();
    expect(count).toBeGreaterThan(0); // beforeEach guarantees at least one ingredient/recipe

    // Find the row that contains OUR seeded recipe. Easiest: open the row whose data-purchase-item-id
    // matches what beforeEach picked. If we reused an existing recipe, just use the first row.
    await rows.first().click();
    await page.waitForTimeout(200);
    const allocRows = rows.first().locator('.recipe-allocation-row');
    const allocCount = await allocRows.count();
    expect(allocCount).toBeGreaterThan(0);

    const slider = allocRows.first().locator('.recipe-slider');
    const chip = allocRows.first().locator('.recipe-pct-chip');
    const putPromise = page.waitForRequest((req) => req.method() === 'PUT' && req.url().includes('/api/v1/inventory/recipes/'));

    await slider.evaluate((el, v) => {
      el.value = String(v);
      el.dispatchEvent(new Event('input', { bubbles: true }));
      el.dispatchEvent(new Event('change', { bubbles: true }));
    }, 45);

    const putReq = await putPromise;
    expect(JSON.parse(putReq.postData() || '{}').usage_pct).toBe(45);
    await expect(chip).toContainText('45%');
    await page.waitForTimeout(800);

    // Switch to Stock tab — Recipes state should be set aside but persisted server-side.
    await page.click('#t2');
    await page.waitForTimeout(400);

    // Switch back to Recipes — fresh load from server.
    await page.click('#t4');
    await page.waitForTimeout(800);

    // Re-expand the same ingredient row (state was cleared because we switched tabs).
    const rowsAfter = page.locator('.recipe-ingredient-row');
    await rowsAfter.first().click();

- **Date range:** `from`/`to` are inclusive YYYY-MM-DD dates interpreted in `America/Chicago`. For Monday–Sunday "May 25–31, 2026" pass `from=2026-05-25&to=2026-05-31`.

| Field | Type | Notes |
|-------|------|-------|
| `from` | string | Echo of input `from` (YYYY-MM-DD) |
| `to` | string | Echo of input `to` |
| `menu_items` | array | One entry per menu_item that has at least one recipe row. Items with no recipes do NOT appear (their spend flows to `unallocated_cogs`). |
| `menu_items[].menu_item_id` | string (UUID) | HQ-side internal ID; correlates to menu_items.id |

Error envelopes mirror Phase 21 EXACTLY (D-18). Same `Content-Type: application/json` body shape:

| Status | Body | Cause |
|--------|------|-------|
| 400 | `{"error":"from must be YYYY-MM-DD"}` | `from` missing or malformed |
| 400 | `{"error":"to must be YYYY-MM-DD"}` | `to` missing or malformed |
| 400 | `{"error":"from must be <= to"}` | `from > to` (lexicographic compare, safe for YYYY-MM-DD) |
| 401 | `{"error":"unauthorized"}` | `Authorization: Bearer` header missing or wrong |
| 500 | `{"error":"internal_error"}` | DB or server-side failure (HQ logs the specifics; the wire response is intentionally generic) |
| 503 | `{"error":"service_token_not_configured"}` | `HQ_INVENTORY_SERVICE_TOKEN` env var is unset on the HQ side — fail-closed |

## 2. Behavior notes

- **No completeness gate field.** Unlike Phase 21, this endpoint does NOT return a `completeness` block (D-15). Drift surfaces in-app (Recipes tab banner) and via Cliq alerts (`purchaseandinventory` channel), NOT via the HTTP response. Sales-processor consumers SHOULD NOT block report generation on a `menu-cogs` field.
- **Idempotent + cacheable.** Same `(from, to)` produces the same response. `Cache-Control: private, max-age=3600` is sent on 200 responses. Sales-processor MAY cache; HQ does NOT cache server-side.
- **Reconciliation with `period-summary`.** Use Phase 21's endpoint for the period totals; use `menu-cogs` for the per-menu-item breakdown. They share the same tax-inclusive math basis (D-10/D-11), so totals reconcile to within sub-cent rounding.
- **Zero-sales menu items still appear** when they have recipe rows but no `daily_menu_sales` in the window (D-07). In that case `units_sold = 0` and `ingredient_cost_per_unit = null` — render as `—`.
