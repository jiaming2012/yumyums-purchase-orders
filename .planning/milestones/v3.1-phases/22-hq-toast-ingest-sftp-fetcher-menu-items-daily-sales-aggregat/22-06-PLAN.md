```js
async function loadMenu(){
  showSkeleton('menu-list');
  try{
    var since=new Date(Date.now()-7*24*60*60*1000).toISOString().slice(0,10);
    var data=await api('/api/v1/inventory/menu-items?since='+since);
    MENU_DATA=data||[];
    renderMenu();
  }catch(e){
    if(e.message==='unauthorized')return;
    showInlineError('menu-list','Couldn’t load menu items.','loadMenu()');
  }
}

function renderMenu(){
  var list=document.getElementById('menu-list');
  if(!list)return;
  if(!MENU_DATA.length){
    list.innerHTML='<div class="empty"><h3 style="font-size:15px;font-weight:500;margin:0 0 6px">No menu items</h3><p>Toast ingest has not populated the menu yet. Run <code>sync-toast</code> or wait for the next 12h cycle.</p></div>';
    return;
  }
  var html='';
  MENU_DATA.forEach(function(m){
    var subgroup=m.menu_subgroup?' &middot; '+escHtml(m.menu_subgroup):'';
    html+='<div class="stock-item" style="display:flex;align-items:center;gap:12px;padding:12px;border-bottom:1px solid var(--brd)">'+
      '<div style="flex:1;min-width:0">'+
        '<div class="stock-item-name" style="font-weight:500">'+escHtml(m.name)+'</div>'+
        '<div style="font-size:12px;color:var(--mut);margin-top:2px">'+escHtml(m.menu_group||'')+subgroup+' &middot; last sold '+formatDate(m.last_seen)+'</div>'+
      '</div>'+
      '<div style="text-align:right">'+
        '<div style="font-weight:500">'+(m.units_sold_this_week||0)+'</div>'+
        '<div style="font-size:11px;color:var(--mut)">this week</div>'+
      '</div>'+
    '</div>';
  });
  list.innerHTML=html;
}
```
